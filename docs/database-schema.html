⚙️ Database Design Rules
Critical constraints enforced across the schema
💰
Financial Precision: All monetary columns use DECIMAL(15,2) — supports up to ₹9,99,99,99,99,999.99. Never use FLOAT or DOUBLE for money.
🔗
Referential Integrity: All foreign keys enforced at DB level. Child records CASCADE DELETE when parent project is deleted (NFR20).
📝
Audit Trail: All cost entry changes logged in cost_entry_audit with previous_amount, new_amount, changed_by, timestamp, and reason (NFR19).
📅
Period Format: All period columns use "YYYY-MM" string format (e.g., "2023-10"). Not a date type — represents a month, not a specific day.
🔢
Auto-creation: When a project is created, all 8 cost_categories rows are automatically inserted for that project.
🗂️
Naming: All tables use snake_case plural names. All columns use snake_case. Prisma @@map() bridges to camelCase in TypeScript.
⏰
Timestamps: Every table has created_at (DEFAULT now()). Mutable tables also have updated_at (auto-updated by Prisma).
🔒
Sessions: express-session + connect-pg-simple stores sessions in a PostgreSQL session table (auto-managed, not in Prisma schema).

model User {
  id            Int       @id @default(autoincrement())
  username      String    @unique
  passwordHash  String    @map("password_hash")
  displayName   String    @map("display_name")
  email         String    @unique
  role          Role
  businessUnit  String?   @map("business_unit")
  isActive      Boolean   @default(true) @map("is_active")
  createdAt     DateTime  @default(now()) @map("created_at")
  updatedAt     DateTime  @updatedAt @map("updated_at")
  managedProjects  Project[]       @relation("ProjectManager")
  costEntries      CostEntry[]
  healthUpdates    HealthUpdate[]
  reviewNotes      ReviewNote[]
  @@map("users")
}

model Project {
  id              Int            @id @default(autoincrement())
  code            String         @unique
  name            String
  client          String
  contractValue   Decimal        @db.Decimal(15, 2) @map("contract_value")
  startDate       DateTime       @map("start_date") @db.Date
  endDate         DateTime       @map("end_date") @db.Date
  status          ProjectStatus  @default(PROPOSAL)
  businessUnit    String         @map("business_unit")
  description     String?
  managerId       Int            @map("manager_id")
  createdAt       DateTime       @default(now()) @map("created_at")
  updatedAt       DateTime       @updatedAt @map("updated_at")
  manager           User                @relation("ProjectManager", fields: [managerId], references: [id])
  paymentMilestones PaymentMilestone[]
  allocations       ProjectAllocation[]
  costCategories    CostCategory[]
  costEntries       CostEntry[]
  healthUpdates     HealthUpdate[]
  risks             Risk[]
  scopeCreepEntries ScopeCreepEntry[]
  reviewNotes       ReviewNote[]
  @@map("projects")
}

model CostEntry {
  id             Int          @id @default(autoincrement())
